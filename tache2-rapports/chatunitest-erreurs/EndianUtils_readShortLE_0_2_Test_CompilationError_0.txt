package org.apache.tika.io;

import java.io.ByteArrayInputStream;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;

public class EndianUtils_readShortLE_0_2_Test {

    private InputStream inputStream;

    private EndianUtils endianUtils;

    @BeforeEach
    public void setUp() {
        inputStream = mock(InputStream.class);
        endianUtils = new EndianUtils();
    }

    @Test
    public void testReadShortLE() throws IOException {
        // Test case 1: Normal case with two bytes
        when(inputStream.read()).thenReturn(0x01, 0x02);
        assertEquals(0x0201, endianUtils.readShortLE(inputStream));
        // Test case 2: Normal case with two bytes in reverse order
        when(inputStream.read()).thenReturn(0x02, 0x01);
        assertEquals(0x0102, endianUtils.readShortLE(inputStream));
        // Test case 3: Normal case with two bytes in reverse order with negative values
        when(inputStream.read()).thenReturn(0xFF, 0xFF);
        assertEquals(0xFF00, endianUtils.readShortLE(inputStream));
        // Test case 4: BufferUnderrunException when not enough bytes are available
        when(inputStream.read()).thenReturn(0x01);
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readShortLE(inputStream));
    }

    @Test
