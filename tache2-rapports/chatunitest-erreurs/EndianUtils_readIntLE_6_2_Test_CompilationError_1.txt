public class EndianUtils_readIntLE_6_2_Test {

        endianUtils = new EndianUtils();

    @Test
    public void testReadIntLENormal() throws IOException {
        // Arrange
        int expectedValue = 12345678;
        byte[] bytes = new byte[] { (byte) 0x78, (byte) 0x56, (byte) 0x34, (byte) 0x12 };
        when(inputStream.read()).thenReturn(bytes[0], bytes[1], bytes[2], bytes[3]);
        // Act
