        when(mockInputStream.read()).thenReturn(0x12, 0x34);
        assertThrows(BufferUnderrunException.class, () -> invokePrivateMethod(EndianUtils.class, "readUIntBE", mockInputStream));
    }

    private Object invokePrivateMethod(Class<?> clazz, String methodName, Object... args) throws Exception {
        java.lang.reflect.Method method = clazz.getDeclaredMethod(methodName, args.getClass());
        method.setAccessible(true);
        return method.invoke(clazz.getDeclaredConstructor().newInstance(), args);
    }
