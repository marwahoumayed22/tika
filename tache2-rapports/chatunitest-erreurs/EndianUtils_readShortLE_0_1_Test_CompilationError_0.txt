package org.apache.tika.io;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.apache.tika.exception.BufferUnderrunException;
import org.apache.tika.exception.TikaException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;

public class EndianUtils_readShortLE_0_1_Test {

    private EndianUtils endianUtils;

    private InputStream inputStream;

    @BeforeEach
    public void setUp() {
        endianUtils = new EndianUtils();
        inputStream = mock(InputStream.class);
    }

    @Test
    public void testReadShortLE_NormalOperation() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenReturn(0x01, 0x02);
        // Act
        short result = endianUtils.readShortLE(inputStream);
        // Assert
        assertEquals(0x0201, result);
    }

    @Test
    public void testReadShortLE_BufferUnderrun() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenReturn(0x01);
        // Act & Assert
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readShortLE(inputStream));
    }

    @Test
    public void testReadShortLE_IOException() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenThrow(new IOException());
        // Act & Assert
        assertThrows(IOException.class, () -> endianUtils.readShortLE(inputStream));
    }

    @Test
    public void testReadShortLE_TikaException() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenThrow(new TikaException());
        // Act & Assert
        assertThrows(TikaException.class, () -> endianUtils.readShortLE(inputStream));
    }

    @Test
    public void testReadShortLE_InvalidInput() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenReturn(-1, 0x02);
        // Act & Assert
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readShortLE(inputStream));
    }

    @Test
    public void testReadShortLE_InvalidInput2() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenReturn(0x01, -1);
        // Act & Assert
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readShortLE(inputStream));
    }

    @Test
    public void testReadShortLE_InvalidInput3() throws IOException, BufferUnderrunException {
        // Arrange
        when(inputStream.read()).thenReturn(-1, -1);
        // Act & Assert
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readShortLE(inputStream));
    }
}
