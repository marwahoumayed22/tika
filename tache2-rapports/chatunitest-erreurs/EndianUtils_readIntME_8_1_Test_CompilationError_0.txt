package org.apache.tika.io;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.apache.tika.exception.BufferUnderrunException;
import org.apache.tika.io.EndianUtils;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.apache.tika.exception.TikaException;

public class EndianUtils_readIntME_8_1_Test {

    private EndianUtils endianUtils;

    private InputStream inputStream;

    @BeforeEach
    public void setUp() {
        endianUtils = new EndianUtils();
        inputStream = mock(InputStream.class);
    }

    @Test
    public void testReadIntME_NormalCase() throws IOException {
        // Arrange
        int[] expectedValues = { 0x12345678 };
        when(inputStream.read()).thenReturn(expectedValues[0], expectedValues[1], expectedValues[2], expectedValues[3]);
        // Act
        int result = endianUtils.readIntME(inputStream);
        // Assert
        assertEquals(expectedValues[0] << 24 | expectedValues[1] << 16 | expectedValues[2] << 8 | expectedValues[3], result);
    }

    @Test
    public void testReadIntME_EdgeCase_ByteOrder() throws IOException {
        // Arrange
        int[] expectedValues = { 0x78, 0x56, 0x34, 0x12 };
        when(inputStream.read()).thenReturn(expectedValues[0], expectedValues[1], expectedValues[2], expectedValues[3]);
        // Act
        int result = endianUtils.readIntME(inputStream);
        // Assert
        assertEquals(expectedValues[0] << 24 | expectedValues[1] << 16 | expectedValues[2] << 8 | expectedValues[3], result);
    }

    @Test
    public void testReadIntME_BufferUnderrunException() throws IOException {
        // Arrange
        when(inputStream.read()).thenReturn(-1);
        // Act and Assert
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readIntME(inputStream));
    }

    @Test
    public void testReadIntME_BufferUnderrunException_WithValues() throws IOException {
        // Arrange
        int[] expectedValues = { 0x12345678 };
        when(inputStream.read()).thenReturn(expectedValues[0], expectedValues[1], expectedValues[2], -1);
        // Act and Assert
        assertThrows(BufferUnderrunException.class, () -> endianUtils.readIntME(inputStream));
    }
}
